add_filter( 'woocommerce_product_tabs', 'Questions_Answers' );
function Questions_Answers( $tabs ) {	
	$tabs_title = "Questions & Answers";

	$tabs['Questions_Answers'] = array(
		'title' 	=> __( $tabs_title, 'woocommerce' ),
		'priority' 	=> 20,
		'callback' 	=> 
'Questions_Answers_tab_content'
	);
	return $tabs;
}
function Questions_Answers_tab_content() { 
$QA_raw = (get_post_meta(get_the_ID(), 'qa', false));
if ($QA_raw){	
foreach ($QA_raw as $key => $value) {	?>
	<details>
	  <summary><?= explode("\n",$value)[0]; ?></summary>
	 <?php echo join(" ",array_slice(explode("\n",$value),1));	 ?>
	</details>			
<?php } } ?>
<style>

  --details-main: #ebeae9;  
}

details p {
    color: #414141;
}

  transition: max-height 1s ease;
  overflow: hidden;
}

details[open] {

}

}

}

details:not([open])
{max-height: 5rem;}
</style>

  }
